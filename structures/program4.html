Question4:
Enter student details like roll name, fee, DOB and print all the detail by ascending order of roll number ?
C program/Syntax
#include <stdio.h>

#include

struct student
{
    int roll;
    char name[30];
    float fee;
    char dob[15];
};

void main()
{
    int i, j;
    struct student stu[50], temp;

    for(i=0;i<50;i++)
    {
        printf("\nEnter details of Student %d\n",i+1);

        printf("Enter Roll Number = ");
        scanf("%d",&stu[i].roll);

        printf("Enter Name = ");
        scanf("%s",stu[i].name);

        printf("Enter Fee = ");
        scanf("%f",&stu[i].fee);

        printf("Enter DOB (DD-MM-YYYY) = ");
        scanf("%s",stu[i].dob);
    }

    for(i=0;i<49;i++)
    {
        for(j=0;j<49-i;j++)
        {
            if(stu[j].roll > stu[j+1].roll)
            {
                temp=stu[j];
                stu[j]=stu[j+1];
                stu[j+1]=temp;
            }
        }
    }

    printf("\n--------------------------------------------------\n");
    printf("Roll\tName\tFee\t\tDOB\n");
    printf("--------------------------------------------------\n");

    for(i=0;i<50;i++)
    {
       printf("%d\t%s\t%f\t\t%s\n",stu[i].roll,stu[i].name,stu[i].fee,stu[i].dob);
    }
}
