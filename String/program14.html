Question14:
Enter the string from user and enter a character , it is exist or not ?
C program/Syntax
#include <stdio.h>

void main()
{
    char str[20], ch;
    int i, flag = 0;

    printf("Enter a string: ");
    scanf("%s", str);

    printf("Enter a character: ");
    scanf(" %c", &ch);

    for(i = 0; str[i] != '\0'; i++)
    {
        if(str[i] == ch)
        {
            flag = 1;
            break;
        }
    }

    if(flag == 1)
        printf("Character exists in the string");
    else
        printf("Character does not exist in the string");
}

Enter a string: computer
Enter a character: p
Output 
                    
Character exists in string
