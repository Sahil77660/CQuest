Question15:
Enter the string from user and count all vowels ?
C program/Syntax
#include <stdio.h>

void main()
{
    char str[20];
    int i, count = 0;

    printf("Enter a string: ");
    scanf("%s", str);

    for(i = 0; str[i] != '\0'; i++)
    {
        if(str[i] == 'a' || str[i] == 'e' || str[i] == 'i' ||
           str[i] == 'o' || str[i] == 'u' || str[i] == 'A' ||
           str[i] == 'E' || str[i] == 'I' || str[i] == 'O' ||
           str[i] == 'U')
        {
            count++;
        }
    }

    printf("Total vowels = %d", count);
}
Input 
                    
Enter a string: education
Output 
                    
Total vowels = 5
