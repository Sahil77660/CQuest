Question1:
Enter a string from user and convert in uppercase ?
C program/Syntax
#include <stdio.h>

void main()
{
    char ch1[20];

    Printf("Enter the name =")
    scanf("%d",ch1);

    for (i = 0; ch1[i] != '\0'; i++)
    {

        if (ch1[i] >= 'a' && ch1[i] <= 'z')
        {
            ch1[i] = ch1[i] - 32;
        }

    }

}

Enter the name = ram
