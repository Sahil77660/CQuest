Question5:
Enter a string from user and convert into toggle case ?
C program/Syntax
#include <stdio.h>

void main()
{
    char ch1[100];
    int i;

    printf("Enter the string = ");
    gets(ch1);

    for (i = 0; ch1[i] != '\0'; i++)
    {
        if (ch1[i] >= 'a' && ch1[i] <= 'z')
        {
            ch1[i] = ch1[i] - 32;
        }
        else if (ch1[i] >= 'A' && ch1[i] <= 'Z')
        {
            ch1[i] = ch1[i] + 32;
        }
    }

    printf("Toggle case = %s", ch1);
}

Enter the string = heLlo worLD
