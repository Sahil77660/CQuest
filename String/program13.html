Program 13
Question13:
Enter two string from user and print difference each character ?
C program/Syntax
#include <stdio.h>

void main()
{
    char str1[20], str2[20];
    int i, diff;

    printf("Enter first string: ");
    scanf("%s", str1);

    printf("Enter second string: ");
    scanf("%s", str2);

    for(i = 0; str1[i] != '\0' && str2[i] != '\0'; i++)
    {
        diff = str1[i] - str2[i];
        printf("%c - %c = %d\n", str1[i], str2[i], diff);
    }
}
Input 
                    
Enter first string: ABC
Enter second string: ACE
Output 
                    
A - A = 0
B - C = -1
C - E = -2
