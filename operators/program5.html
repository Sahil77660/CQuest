Question5:
Write a program to enter 2 numbers from user add the number and show the answer , again take 2 number from user subtract the number display the answer ,again take 2 number from user multiply the number display the answer and ,again take 2 number from the user divide the number display the answer ,again take 2 number find the remainder and display the answer ?
C program/Syntax
#include <stdio.h>

void main()
{
    int num1,num2;

    printf("Enter number 1 for addition: ");
    scanf("%d",&num1);
    printf("Enter number 2 for addition: ");
    scanf("%d",&num2);
    printf("Addition = %d\n",num1+num2);

    printf("\nEnter number 1 for subtraction: ");
    scanf("%d",&num1);
    printf("Enter number 2 for subtraction: ");
    scanf("%d",&num2);
    printf("Subtraction = %d\n",num1-num2);

    printf("\nEnter number 1 for multiplication: ");
    scanf("%d",&num1);
    printf("Enter number 2 for multiplication: ");
    scanf("%d",&num2);
    printf("Multiplication = %d\n",num1*num2);

    printf("\nEnter number 1 for division: ");
    scanf("%d",&num1);
    printf("Enter number 2 for division: ");
    scanf("%d",&num2);
    printf("Division = %d\n",num1/num2);

    printf("\nEnter number 1 for remainder: ");
    scanf("%d",&num1);
    printf("Enter number 2 for remainder: ");
    scanf("%d",&num2);
    printf("Remainder = %d",num1%num2);

}
Input 
                    
Enter number 1 for addition: 10
Enter number 2 for addition: 5

Enter number 1 for subtraction: 10
Enter number 2 for subtraction: 5
Enter number 1 for multiplication: 10
Enter number 2 for multiplication: 5
Enter number 1 for division: 10
Enter number 2 for division: 5
Enter number 1 for remainder: 10
Enter number 2 for remainder: 3
Output 
                    
Addition = 15
Subtraction = 5
Multiplication = 50
Division = 2
Remainder = 1
