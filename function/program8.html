Question8:
Making Armstrong program using local and global variable ?
C program/Syntax
Using local variable
            
#include <stdio.h>

void armstrong()
{
    int num, original, r, sum=0;

    printf("Enter a number: ");
    scanf("%d",&num);

    original=num;

    while(num>0)
    {
        r=num%10;
        sum=sum+(r*r*r);
        num=num/10;
    }

    if(original==sum)
        printf("Armstrong number");
    else
        printf("Not an Armstrong number");
}

void main()
{
    armstrong();
}
