Question4:
Making Factorial program using local and global variable ?
C program/Syntax
Using local variable
            
#include <stdio.h>

void factorial()
{
    int num, i, fact=1;

    printf("Enter a number: ");
    scanf("%d",&num);

    for(i=1; i<=num; i++)
    {
        fact=fact*i;
    }

    printf("Factorial = %d",fact);
}

void main()
{
    factorial();
}
