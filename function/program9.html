Question9:
Making Fibonacci program using local and global variable ?
C program/Syntax
Using local variable
            
#include <stdio.h>

void fibonacci()
{
    int num, a=0, b=1, c, i;

    printf("Enter number of terms: ");
    scanf("%d",&num);

    for(i=1; i<=num; i++)
    {
        printf("%d ",a);

        c=a+b;
        a=b;
        b=c;
    }
}

void main()
{
    fibonacci();
}
