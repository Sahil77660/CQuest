Question1:
Enter an array from user and display all the value ?
C program/Syntax
#include <stdio.h>

void main()
    {
        int arr[5],i;
        
        for(i=0;i<=4;i++)
        {
            printf("Enter the number %d =",i);
            scanf("%d",&arr[i]);
        }
        printf("Array elements are...\n");

        for(i=0;i<=4;i++)
        {
            printf("%d\n",arr{i});
        }

    }
Input 
                    
 Enter the number 1 =10
 Enter the number 2 =60
 Enter the number 3 =20
 Enter the number 4 =104
 Enter the number 5 =107
Output 
                    
Array elements are...
10
60
20
104
