Program 15
Question15:
Enter one digit number from user and print in words ?
C program/Syntax
#include <stdio.h>
void main()
{
    int num;

    printf("Enter a one digit number = ");
    scanf("%d",&num);

    switch(num)
    {
        case 0: printf("Zero");
                break;
        case 1: printf("One");
                break;
        case 2: printf("Two");
                break;
        case 3: printf("Three");
                break;
        case 4: printf("Four");
                break;
        case 5: printf("Five");
                break;
        case 6: printf("Six");
                break;
        case 7: printf("Seven");
                break;
        case 8: printf("Eight");
                break;
        case 9: printf("Nine");
                break;
        default: printf("Invalid number");
    }
}
Input 
                    
Enter a one digit number = 8
Output 
                    
Eight
