Program 18
Question18:
Enter a character from user and it is in upper case, lower case, special characters or numbers ?
C program/Syntax
#include <stdio.h>

void main()
{
    char ch1;
    printf("Enter the character =");
    scanf("%c",&ch1);

    if(ch1>='a' && ch1<='z')
    {
        printf("Lower case alphabet");
    }
    else if(ch1>='A' && ch1<='Z')
    {
        printf("Upper case alphabet");
    }
    else if(ch1>='0' && ch1<='9')
    {
        printf("Number");
    }
    else{
        printf("Special character");
    }

}
