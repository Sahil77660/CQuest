Question12:
Enter a number from the user if the user enter 1 then you will perform addition operation by entering 2 number from user add it and display answer. if user enter 2 then subtract or if user enter 3 then multiply or if user enter 4 then divide or if user enter 5 then modulo or if any other number then print invalid .if user want to quit the program then user will enter Q. ?
C program/Syntax

#include <stdio.h>

void main()
{
    int num1, num2;
    char choice;

    while(1)
    {
        printf("\n1. Addition");
        printf("\n2. Subtraction");
        printf("\n3. Multiplication");
        printf("\n4. Division");
        printf("\n5. Modulo");
        printf("\nQ. Quit");

        printf("\n\nEnter your choice: ");
        scanf(" %c",&choice);

        if(choice=='Q' || choice=='q')
        {
            printf("Program terminated.");
            break;
        }

        if(choice>='1' && choice<='5')
        {
            printf("Enter first number: ");
            scanf("%d",&num1);

            printf("Enter second number: ");
            scanf("%d",&num2);
        }

        switch(choice)
        {
            case '1':
                printf("Addition = %d",num1+num2);
                break;

            case '2':
                printf("Subtraction = %d",num1-num2);
                break;

            case '3':
                printf("Multiplication = %d",num1*num2);
                break;

            case '4':
                if(num2!=0)
                    printf("Division = %d",num1/num2);
                else
                    printf("Cannot divide by zero.");
                break;

            case '5':
                if(num2!=0)
                    printf("Modulo = %d",num1%num2);
                else
                    printf("Cannot perform modulo by zero.");
                break;

            default:
                printf("Invalid choice.");
        }
    }
}

1. Addition

5. Modulo
Q. Quit

Enter second number: 10
