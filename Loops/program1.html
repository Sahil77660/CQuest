Program 01
Question1:
Enter a number from the user and find out the binary of that number ?
C program/Syntax
#include <stdio.h>

void main(){
    int num,binary[32];
    int i=0;
    printf("enter a number:");
    scanf("%d",&num);
    if(num==0){
        printf("binary :0");}
    while (num>0){
        binary[i]=num%2;
        i++;
        num=num/2;}
    printf("binary: ");
    for(i=i-1;i>=0;i--){
        printf("%d",binary[i]);}
}
Input 
                    
enter a number:2026
Output 
                    
enter a number:2026
binary: 11111101010
