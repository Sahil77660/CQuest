Question15:
Enter a number from user and print first digit of number ?
C program/Syntax

#include <stdio.h>

void main()
{
    int num;

    printf("Enter a number: ");
    scanf("%d",&num);

    while(num>=10)
    {
        num=num/10;
    }

    printf("First digit = %d",num);
}
Input 
                    
Enter a number: 4587
Output 
                    
First digit = 4
